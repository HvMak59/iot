async findAssetOrgIdMap(assetIds: string[]) {
        const assets = await this.repo.find({
            select: {
                id: true,
                orgId: true,
            },
            where: {
                id: In(assetIds),
            },
        });

        return new Map(
            assets.map(asset => [asset.id, asset.orgId]),
        );
    }
